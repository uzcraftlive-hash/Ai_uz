# install: pip install faker from faker import Faker import json import random fake = Faker(['uz_UZ', 'ru_RU', 'en_US']) # 500,000 ta Q&A yarating data = [] topics = ['uzbekistan', 'tarix', 'fan', 'matematika', 'texnologiya', 'sog\'lik', 'sport'] for i in range(500000): topic = random.choice(topics) question = fake.sentence() answer = fake.text(max_nb_chars=200) data.append({ 'id': i, 'topic': topic, 'question': question, 'answer': answer, 'timestamp': fake.iso8601() }) if i % 10000 == 0: print(f"Generated {i}/500000...") # JSON faylga saqlang with open('ai_database_100mb.json', 'w', encoding='utf-8') as f: json.dump(data, f, ensure_ascii=False, indent=2) print("✅ Yaratildi: ai_database_100mb.json (~100MB)")
